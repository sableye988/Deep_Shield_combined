# 경로 지정 
$OUT = "C:\Users\정은성\Desktop\deepfake\out"  <= 본인 경로로 하시면 됩니다
New-Item -ItemType Directory -Path $OUT -Force | Out-Null

# 워터마크 삽입 코드
curl.exe -sS --fail-with-body -w "`nHTTP %{http_code}`n" `
  "http://127.0.0.1:8000/embed_blind" `
  -F "host=@C:\Users\정은성\Desktop\deepfake\face.jpg;type=image/jpeg" `
  -F "wm=@C:\Users\정은성\Desktop\deepfake\hanshin.png;type=image/png" `
  -F "wavelet=db2" -F "level=1" -F "bands=HL,LH" -F "delta=4.0" -F "repeat=1" `
  -o "$OUT\face_marked_blind.png"
